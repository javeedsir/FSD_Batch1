Express.js  App and Mysql setup
1. mkdir  my-express-app
2. >my-express-app> npm install express
3. > my-express-app> npm install mysql2


Note: we should have NodeJS server installed in our machines before installing expressJS.

index.js(create this file and write below code)


import  express from 'express'
import mysql from 'mysql2/promise'

const app =  express();

const PORT = 3000;

app.listen(PORT , ()=>{

        console.log(`Server Running on PORT ${PORT}`);

} );


const pool = mysql.createPool(
        {

            host:'localhost',
            user:'root',
            password:'admin',
            database:'upgrad_db'

        }

);




app.get('/employees' , async (req,res)=>{


        try{

            const  [rows]  =  await  pool.query("select * from Employees");

                res.json(rows);
        }
        catch(error){

                res.status(500).json({message:"Database Query failed"})


        }


});
